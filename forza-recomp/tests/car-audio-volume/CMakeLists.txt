cmake_minimum_required(VERSION 3.25)
project(car_audio_volume_checks LANGUAGES CXX)
enable_testing()
foreach(tu 0 4)
  add_executable(car_audio_volume_${tu} car_audio_volume_test.cpp ../../src/hooks/car_audio_volume.cpp)
  target_include_directories(car_audio_volume_${tu} PRIVATE stubs ../../src)
  target_compile_features(car_audio_volume_${tu} PRIVATE cxx_std_20)
  target_compile_definitions(car_audio_volume_${tu} PRIVATE FORZA_TITLE_UPDATE=${tu} _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    target_compile_options(car_audio_volume_${tu} PRIVATE /W4 /WX /UNDEBUG)
  else()
    target_compile_options(car_audio_volume_${tu} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME car_audio_volume_${tu} COMMAND car_audio_volume_${tu})
endforeach()
